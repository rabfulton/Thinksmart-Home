cmake_minimum_required(VERSION 3.21)
project(ThinkSmartAudioLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core)
set(MICRO_FRONTEND "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/microfrontend")
set(MICRO_FRONTEND_LIB "${MICRO_FRONTEND}/tensorflow/lite/experimental/microfrontend/lib")
set(MICRO_FRONTEND_SOURCES)
foreach(name kiss_fft_int16 fft fft_util filterbank filterbank_util frontend frontend_util log_lut log_scale log_scale_util noise_reduction noise_reduction_util pcan_gain_control pcan_gain_control_util window window_util)
    list(APPEND MICRO_FRONTEND_SOURCES "${MICRO_FRONTEND_LIB}/${name}.cc")
endforeach()
add_library(thinksmart-micro STATIC microdetector.cpp microdetector.h ${MICRO_FRONTEND_SOURCES} "${MICRO_FRONTEND}/kissfft/kiss_fft.cc" "${MICRO_FRONTEND}/kissfft/tools/kiss_fftr.cc")
target_include_directories(thinksmart-micro PRIVATE "${MICRO_FRONTEND}" "${MICRO_FRONTEND}/kissfft")
target_compile_definitions(thinksmart-micro PRIVATE FIXED_POINT=16)
target_link_libraries(thinksmart-micro PUBLIC Qt6::Core)
add_executable(thinksmart-micro-check micro-check.cpp)
target_link_libraries(thinksmart-micro-check PRIVATE thinksmart-micro)
# Select pkg-config's target architecture explicitly for cross compilation.
if(CMAKE_CROSSCOMPILING)
    set(ENV{PKG_CONFIG_LIBDIR} "/usr/lib/aarch64-linux-gnu/pkgconfig:/usr/share/pkgconfig")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(ALSA REQUIRED IMPORTED_TARGET alsa)
pkg_check_modules(APM REQUIRED IMPORTED_TARGET webrtc-audio-processing-1)
pkg_check_modules(SPEEX REQUIRED IMPORTED_TARGET speexdsp)
add_executable(thinksmart-audio-lab lab.cpp aec.cpp aec.h)
target_link_libraries(thinksmart-audio-lab PRIVATE Qt6::Core PkgConfig::ALSA PkgConfig::APM PkgConfig::SPEEX)
target_compile_options(thinksmart-audio-lab PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    add_test(NAME aec-signal-check COMMAND thinksmart-audio-lab --self-test)
    add_executable(ducking-test ducking-test.cpp)
    add_test(NAME ducking-test COMMAND ducking-test)
endif()

add_executable(thinksmart-voice-audio stream.cpp aec.cpp aec.h pcm.h)
target_link_libraries(thinksmart-voice-audio PRIVATE Qt6::Core PkgConfig::ALSA PkgConfig::APM PkgConfig::SPEEX)
target_compile_options(thinksmart-voice-audio PRIVATE -Wall -Wextra -Wpedantic)

set(THINKSMART_SHERPA_ROOT "" CACHE PATH "Pinned Sherpa-ONNX runtime directory")
set(THINKSMART_SHERPA_INCLUDE "" CACHE PATH "Matching C API include directory")
if(THINKSMART_SHERPA_ROOT)
    find_package(Threads REQUIRED)
    pkg_check_modules(PIPEWIRE IMPORTED_TARGET libpipewire-0.3)
    if(PIPEWIRE_FOUND)
        add_executable(thinksmart-shared-audio shared.cpp aec.cpp)
        target_include_directories(thinksmart-shared-audio PRIVATE "${THINKSMART_SHERPA_INCLUDE}")
        target_link_directories(thinksmart-shared-audio PRIVATE "${THINKSMART_SHERPA_ROOT}/lib")
        target_link_libraries(thinksmart-shared-audio PRIVATE thinksmart-micro Qt6::Core PkgConfig::ALSA PkgConfig::APM PkgConfig::SPEEX PkgConfig::PIPEWIRE sherpa-onnx-c-api Threads::Threads)
        set_target_properties(thinksmart-shared-audio PROPERTIES BUILD_RPATH "$ORIGIN/../lib/thinksmart-wakeword;${THINKSMART_SHERPA_ROOT}/lib")
    endif()
    if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
        add_executable(idle-audio-test idle-test.cpp aec.cpp)
        target_include_directories(idle-audio-test PRIVATE "${THINKSMART_SHERPA_INCLUDE}")
        target_link_directories(idle-audio-test PRIVATE "${THINKSMART_SHERPA_ROOT}/lib")
        target_link_libraries(idle-audio-test PRIVATE thinksmart-micro Qt6::Core PkgConfig::ALSA PkgConfig::APM PkgConfig::SPEEX sherpa-onnx-c-api)
        add_test(NAME idle-audio-test COMMAND idle-audio-test "${CMAKE_CURRENT_SOURCE_DIR}/../../build/wakeword")
    endif()
    add_executable(thinksmart-wakeword wake.cpp pcm.h)
    target_include_directories(thinksmart-wakeword PRIVATE "${THINKSMART_SHERPA_INCLUDE}")
    target_link_directories(thinksmart-wakeword PRIVATE "${THINKSMART_SHERPA_ROOT}/lib")
    target_link_libraries(thinksmart-wakeword PRIVATE thinksmart-micro Qt6::Core PkgConfig::ALSA PkgConfig::SPEEX sherpa-onnx-c-api Threads::Threads)
    set_target_properties(thinksmart-wakeword PROPERTIES BUILD_RPATH "$ORIGIN/../lib/thinksmart-wakeword;${THINKSMART_SHERPA_ROOT}/lib")
    target_compile_options(thinksmart-wakeword PRIVATE -Wall -Wextra -Wpedantic)
endif()
